cmake_minimum_required(VERSION 3.24)
project(WaxTranslation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_CLI OFF CACHE BOOL "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(WITH_MKL OFF CACHE BOOL "")
set(WITH_ACCELERATE ON CACHE BOOL "")
set(OPENMP_RUNTIME NONE CACHE STRING "")
set(SPM_ENABLE_SHARED OFF CACHE BOOL "")
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "")
add_subdirectory("${CT2_SOURCE}" ctranslate2 EXCLUDE_FROM_ALL)
add_subdirectory("${SPM_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
add_library(WaxTranslation SHARED Translator.cpp)
target_include_directories(WaxTranslation PRIVATE "${SPM_SOURCE}/src" "${CT2_SOURCE}/include")
target_link_libraries(WaxTranslation PRIVATE ctranslate2 sentencepiece-static)
